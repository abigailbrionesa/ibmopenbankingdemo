Example: Making an API Call

                

                    Security Note: Never include client secrets in frontend code.
                    This example shows the OAuth flow structure. In production, token exchange
                    must happen server-side to protect your client secret.
                

                

// Server-side token exchange endpoint
const tokenResponse = await fetch('/api/token-exchange', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    code: authorizationCode,
    // Client credentials handled securely on server
  })
});

const { access_token } = await tokenResponse.json();

// Call protected API with access token
const apiResponse = await fetch('/api/v1/accounts', {
  headers: {
    'Authorization': `Bearer ${access_token}`
  }
});

const accounts = await apiResponse.json();
console.log(accounts);
